#Parte de la VPN

crypto isakmp policy 10
 encryption des
 hash md5
 authentication pre-share
 group 14
 lifetime 86400
exit


crypto isakmp key Toybot17 address 14.10.2.2

crypto ipsec transform-set TS-VPN esp-des esp-md5-hmac
 mode tunnel
exit

ip access-list extended VPN-TRAFFIC
 permit ip 14.10.10.0 0.0.0.127 14.10.20.0 0.0.0.15
exit

crypto map CMAP 10 ipsec-isakmp
 set peer 14.10.2.2
 set transform-set TS-VPN
 match address VPN-TRAFFIC
exit

interface e0/0
 crypto map CMAP

do wr
